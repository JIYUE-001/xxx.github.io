<script>
    // ===================== DOM节点获取 =====================
    const ta = document.getElementById('ta');
    const tb = document.getElementById('tb');
    const btnCheck = document.getElementById('btnCheck');
    const btnClear = document.getElementById('btnClear');
    const btnCopy = document.getElementById('btnCopy');
    const resultBox = document.getElementById('resultBox');
    const scoreText = document.getElementById('scoreText');
    const viewA = document.getElementById('viewA');
    const viewB = document.getElementById('viewB');
    const toast = document.getElementById('toast');

    // ===================== Toast提示函数 =====================
    function toastShow(msg) {
        toast.innerText = msg;
        toast.style.display = 'block';
        setTimeout(() => {
            toast.style.display = 'none';
        }, 1800);
    }

    // ===================== N-Gram 生成函数（2字滑动窗口） =====================
    function getNgram(text, n = 2) {
        const cleanText = text.replace(/\s/g, '');
        const gramSet = new Set();
        for (let i = 0; i <= cleanText.length - n; i++) {
            const gram = cleanText.slice(i, i + n);
            gramSet.add(gram);
        }
        return gramSet;
    }

    // ===================== 余弦相似度计算 =====================
    function calcCosine(setA, setB) {
        const union = new Set([...setA, ...setB]);
        let dot = 0, magA = 0, magB = 0;
        for (const key of union) {
            const a = setA.has(key) ? 1 : 0;
            const b = setB.has(key) ? 1 : 0;
            dot += a * b;
            magA += a * a;
            magB += b * b;
        }
        if (magA === 0 || magB === 0) return 0;
        return dot / (Math.sqrt(magA) * Math.sqrt(magB));
    }

    // ===================== 文本片段标红渲染 =====================
    function renderMark(text, dupGrams, n = 2) {
        const clean = text.replace(/\s/g, '');
        const markArr = Array(clean.length).fill(false);
        // 标记所有重复ngram覆盖位置
        for (const gram of dupGrams) {
            let pos = 0;
            while (pos <= clean.length - n) {
                if (clean.slice(pos, pos + n) === gram) {
                    for (let i = 0; i < n; i++) {
                        markArr[pos + i] = true;
                    }
                    pos += n;
                } else {
                    pos += 1;
                }
            }
        }
        // 拼接带标签HTML
        let html = '';
        for (let i = 0; i < clean.length; i++) {
            if (markArr[i]) {
                html += `<span class="dup">${clean[i]}</span>`;
            } else {
                html += clean[i];
            }
        }
        return html;
    }

    // ===================== 查重主逻辑 =====================
    function startCheck() {
        const strA = ta.value.trim();
        const strB = tb.value.trim();
        if (!strA || !strB) {
            toastShow('两段文本都不能为空！');
            return;
        }
        const gramA = getNgram(strA, 2);
        const gramB = getNgram(strB, 2);
        // 求交集：共同的短语片段
        const dupGrams = new Set([...gramA].filter(x => gramB.has(x)));
        const sim = calcCosine(gramA, gramB);
        const percent = (sim * 100).toFixed(2);

        // 渲染标红内容
        viewA.innerHTML = renderMark(strA, dupGrams, 2);
        viewB.innerHTML = renderMark(strB, dupGrams, 2);
        scoreText.innerText = `${percent}%`;

        // 分数颜色分级
        scoreText.className = 'score-value';
        if (sim >= 0.6) {
            scoreText.classList.add('score-high');
        } else if (sim >= 0.3) {
            scoreText.classList.add('score-mid');
        } else {
            scoreText.classList.add('score-low');
        }
        resultBox.classList.add('active');
    }

    // ===================== 清空全部 =====================
    function clearAll() {
        ta.value = '';
        tb.value = '';
        resultBox.classList.remove('active');
        toastShow('已清空');
    }

    // ===================== 复制结果 =====================
    async function copyResult() {
        const report = `【本地文本查重报告】
相似度：${scoreText.innerText}
--------------------------
文本A：
${ta.value}
--------------------------
文本B：
${tb.value}`;
        try {
            await navigator.clipboard.writeText(report);
            toastShow('报告复制成功');
        } catch (err) {
            toastShow('复制失败，请手动复制文本');
        }
    }

    // ===================== 绑定点击事件 =====================
    btnCheck.addEventListener('click', startCheck);
    btnClear.addEventListener('click', clearAll);
    btnCopy.addEventListener('click', copyResult);
</script>
